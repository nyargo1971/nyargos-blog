<p>from config.config import ETA_NN005<br />from config.config import NN005_COLLECT_TURN<br />from state.state_nn005 import NN005State</p>
<p><br />class NN005Model:<br />"""<br />NN005 ver3.3（内部ギア搭載モデル）<br />- 外側契約は predict/learn のまま完全維持<br />- 内部に旧NN005ver2 の多段階ギアを封じ込める<br />1. 情報収集モード（初期）<br />2. 通常学習モード（中盤）<br />3. 揺れ検出モード（接点判定）<br />4. 微調整モード（終盤）<br />"""</p>
<p>def __init__(self) -&gt; None:<br /># μ の初期化は state に任せる<br /># 内部ギア用の状態<br />self.turn = 0</p>
<p># ギア切り替え用<br />self.phase = "collect" # "collect" → "normal" → "stable" → "micro"</p>
<p># 揺れ検出用<br />self.last_correct = None<br />self.oscillation_score = 0 # 揺れが多いほど増える</p>
<p># 旧ver2 の「連続正解/不正解」相当<br />self.consecutive_correct = 0<br />self.consecutive_wrong = 0</p>
<p># 内部ステップ倍率（ギアで変わる）<br />self.step_factor = 1.0</p>
<p># ============================================================<br /># μ 初期化<br /># ============================================================<br />def _ensure_initialized(self, state: NN005State, x_norm: float) -&gt; None:<br />if state.mu is None:<br />state.mu = x_norm</p>
<p># ============================================================<br /># 予測（外側契約）<br /># ============================================================<br />def predict(self, x_norm: float, state: NN005State) -&gt; int:<br />self._ensure_initialized(state, x_norm)<br />return 1 if x_norm &gt;= state.mu else -1</p>
<p># ============================================================<br /># ギア切り替えロジック<br /># ============================================================<br />def _update_phase(self):<br />"""<br />turn に応じてギアを切り替える<br />"""<br /># 1. 情報収集モード（最初の 100 ターン）<br />if self.turn &lt; NN005_COLLECT_TURN:<br />self.phase = "collect"<br />return</p>
<p># 2. 通常学習モード（〜1000）<br />if self.turn &lt; 1000:<br />self.phase = "normal"<br />return</p>
<p># 3. 揺れ検出モード（〜5000）<br />if self.turn &lt; 5000:<br />self.phase = "stable"<br />return</p>
<p># 4. 微調整モード（5001〜）<br />self.phase = "micro"</p>
<p># ============================================================<br /># 揺れ検出（接点判定）<br /># ============================================================<br />def _update_oscillation(self, correct: int):<br />"""<br />correct が +1/-1 と揺れるほどスコアが上がる<br />"""<br />if self.last_correct is None:<br />self.last_correct = correct<br />return</p>
<p>if correct != self.last_correct:<br />self.oscillation_score += 1 # 揺れた<br />else:<br />self.oscillation_score = max(0, self.oscillation_score - 0.5)</p>
<p>self.last_correct = correct</p>
<p># ============================================================<br /># 学習（外側契約）<br /># ============================================================<br />def learn(self, x_norm: float, correct: int, state: NN005State) -&gt; None:<br />self._ensure_initialized(state, x_norm)<br />self.turn += 1<br />self._update_phase()<br />self._update_oscillation(correct)</p>
<p>pred = 1 if x_norm &gt;= state.mu else -1</p>
<p># ------------------------------------------------------------<br /># ギア1：情報収集モード（μ を動かさない）<br /># ------------------------------------------------------------<br />if self.phase == "collect":<br />return</p>
<p># ------------------------------------------------------------<br /># ギア2：通常学習モード（中盤）<br /># ------------------------------------------------------------<br />if self.phase == "normal":<br />if pred != correct:<br /># 誤分類 → x_norm に寄せる（回帰型）<br />state.mu += ETA_NN005 * (x_norm - state.mu) * 0.7<br /># 正解 → 少しだけ安定化<br />else:<br />state.mu += ETA_NN005 * (x_norm - state.mu) * 0.7</p>
<p># ------------------------------------------------------------<br /># ギア3：揺れ検出モード（接点判定）<br /># ------------------------------------------------------------<br />elif self.phase == "stable":<br /># 揺れが大きい → 接点 → μ を動かさない<br />if self.oscillation_score &gt;= 5:<br />return</p>
<p># 揺れが少ない → 2交点 → 通常学習<br />if pred != correct:<br />state.mu += ETA_NN005 * (x_norm - state.mu) * 0.7<br />else:<br />state.mu += ETA_NN005 * (x_norm - state.mu) * 0.1</p>
<p># ------------------------------------------------------------<br /># ギア4：微調整モード（終盤）<br /># ------------------------------------------------------------<br />elif self.phase == "micro":<br /># 微調整：極小ステップ<br />if pred != correct:<br />state.mu += ETA_NN005 * (x_norm - state.mu) * 0.05<br />else:<br />state.mu += ETA_NN005 * (x_norm - state.mu) * 0.001</p>
<p># ------------------------------------------------------------<br /># クリップ<br /># ------------------------------------------------------------<br />if state.mu &lt; 0.0:<br />state.mu = 0.0<br />elif state.mu &gt; 1.0:<br />state.mu = 1.0</p>
