<p><br />class NN007Model:<br />def predict(self, x1_norm: float, x2_norm: float, state: NN007State) -&gt; Tuple[int, int]:<br />#------------------------------------------------------------<br /># 予測フェーズ（forward）<br /># ・NN007 は「2つの正規化された x 座標が、学習済みの区間に入っているか」を判定するだけの簡易モデル<br /># ・state.x1_norm, state.x2_norm が “区間の左右端” を表す<br /># ・返り値は (+1 = 内側, -1 = 外側)<br />#------------------------------------------------------------</p>

<p>lo = min(state.x1_norm, state.x2_norm) # 区間の左端<br />hi = max(state.x1_norm, state.x2_norm) # 区間の右端</p>
<p># x1_norm, x2_norm が区間に入っているかを判定<br />s1 = +1 if lo &lt;= x1_norm &lt;= hi else -1<br />s2 = +1 if lo &lt;= x2_norm &lt;= hi else -1<br />return s1, s2</p>
<p>def learn(<br />self,<br />x1_norm: float,<br />x2_norm: float,<br />correct1: int,<br />correct2: int,<br />state: NN007State,<br />) -&gt; None:<br />#------------------------------------------------------------<br /># 学習フェーズ（update）<br /># ・NN007 は「区間を動かす」ことで学習する<br /># ・誤差逆伝播のような複雑な計算は行わず、<br /># “正解に近づく方向へ区間を少しずつ移動させる” というルールベース学習<br /># ・ETA_NN007 が学習率（どれだけ動かすか）<br />#------------------------------------------------------------</p>
<p># 初期化（predict と同じ）<br />if state.x1_norm is None or state.x2_norm is None:<br />state.x1_norm = 0.25<br />state.x2_norm = 0.75</p>
<p>#------------------ 心臓部：区間調整ループ ------------------<br />for x_norm, correct in [(x1_norm, correct1), (x2_norm, correct2)]:<br />lo = min(state.x1_norm, state.x2_norm)<br />hi = max(state.x1_norm, state.x2_norm)</p>
<p>inside = lo &lt;= x_norm &lt;= hi<br />pred = +1 if inside else -1 # 現在の予測</p>
<p># 予測が正解と違う場合のみ、区間を動かす<br />if pred != correct:<br />if correct == +1:<br />#----------------------------------------------<br /># 正解は「内側」なのに、今は外側 → 区間を広げる方向へ動かす<br />#----------------------------------------------<br />if x_norm &lt; lo:<br /># x_norm が左にある → 左端を左へ動かす<br />state.x1_norm -= ETA_NN007 * (lo - x_norm)<br />elif x_norm &gt; hi:<br /># x_norm が右にある → 右端を右へ動かす<br />state.x2_norm += ETA_NN007 * (x_norm - hi)<br />else:<br />#----------------------------------------------<br /># 正解は「外側」なのに、今は内側 → 区間を狭める方向へ動かす<br />#----------------------------------------------<br />if inside:<br />mid = (lo + hi) / 2.0 # 区間の中央<br />if x_norm &lt; mid:<br /># x_norm が左寄り → 左端を右へ押し込む<br />state.x1_norm += ETA_NN007 * (mid - x_norm)<br />else:<br /># x_norm が右寄り → 右端を左へ押し込む<br />state.x2_norm -= ETA_NN007 * (x_norm - mid)</p>
<p>#------------------ 心臓部ここまで --------------------------</p>
<p>#------------------------------------------------------------<br /># 係数 a, b, c のクリップ（安全装置）<br /># ・NN007State には a, b, c があるが、このモデルでは直接使わない<br /># ・値が暴走しないように範囲内に収めるだけ<br />#------------------------------------------------------------<br />state.a = max(NN007_A_MIN, min(NN007_A_MAX, state.a))<br />state.b = max(NN007_B_MIN, min(NN007_B_MAX, state.b))<br />state.c = max(NN007_C_MIN, min(NN007_C_MAX, state.c))</p>
