<p><br />class Environment:<br />def __init__(self, state: EnvironmentState):<br />self.state = state</p>
<p>def initialize_with_input(self, x: float, y: float, r: float) -&gt; None:<br />if r &lt;= 0:<br />raise ValueError("半径 r は正の値である必要があります。")</p>
<p>disc = r * r - y * y<br />if disc &lt; 0:<br />raise ValueError("この真円は x軸と交点を持ちません。y と r を見直してください。")</p>
<p>self.state.target_x = x<br />self.state.target_y = y<br />self.state.target_r = r</p>
<p>self._compute_true_intersections()</p>
<p>def _compute_true_intersections(self) -&gt; None:<br />x0 = self.state.target_x<br />y0 = self.state.target_y<br />r = self.state.target_r</p>
<p>disc = r * r - y0 * y0<br />if disc &lt; 0:<br />self.state.target_left = None<br />self.state.target_right = None<br />return</p>
<p>d = math.sqrt(disc)<br />left = x0 - d<br />right = x0 + d</p>
<p>self.state.target_left = min(left, right)<br />self.state.target_right = max(left, right)</p>

<p>def normalize(self, x_real: float) -&gt; float:<br />return (x_real - HINT_MIN) / (HINT_MAX - HINT_MIN)</p>
<p>def judge(self, x_real: float) -&gt; int:<br />x0 = self.state.target_x<br />y0 = self.state.target_y<br />r = self.state.target_r</p>
<p>dist2 = (x_real - x0) ** 2 + (0 - y0) ** 2<br />return +1 if dist2 &lt;= r * r else -1</p>
